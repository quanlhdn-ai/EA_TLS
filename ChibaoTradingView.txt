f_pivots_filter(float ph, float pl, float prev_max, float prev_min) =>
    float last_high_ = fixnan(ph)
    float max_high_  = not na(pl) ? ph : na(prev_max) ? ph : prev_max
    max_high_ := math.max(max_high_, last_high_)
    float last_low_ = fixnan(pl)
    float min_low_  = not na(ph) ? pl : na(prev_min) ? pl : prev_min
    min_low_ := math.min(min_low_, last_low_)
    float cur_max_high_ = fixnan(max_high_)
    float cur_min_low_  = fixnan(min_low_)
    float pivot_high = not na(ph) or barstate.islast ? (max_high_ == ph ? cur_max_high_ : na) : na
    float pivot_low  = not na(pl) or barstate.islast ? (min_low_ == pl ? cur_min_low_ : na) : na
    [pivot_high, pivot_low, max_high_, min_low_]

draw_ob(float top, float bot, bool isBull, int origin_bar_idx, float imb_t, float imb_b) =>
    if EnableOB
        b = box.new(origin_bar_idx, top, bar_index + 10, bot, border_color=na, bgcolor=isBull ? c_BuyOB : c_SellOB)
        label imb_lbl = label(na)
        line imb_line_t = line(na)
        line imb_line_b = line(na)
        bool has_imb = (imb_t > imb_b)
        if has_imb
            imb_lbl := label.new(origin_bar_idx + (bar_index + 10 - origin_bar_idx) / 2, (imb_t + imb_b) / 2, "Imbalance", color=color.new(color.white, 100), textcolor=color.black, style=label.style_label_center, size=size.small)
            imb_line_t := line.new(origin_bar_idx, imb_t, bar_index + 10, imb_t, color=color.new(color.black, 50), style=line.style_dashed, width=1)
            imb_line_b := line.new(origin_bar_idx, imb_b, bar_index + 10, imb_b, color=color.new(color.black, 50), style=line.style_dashed, width=1)
        l = label.new(bar_index + 10, (top + bot) / 2, isBull ? "Buy OB" : "Sell OB", color=color.new(color.white, 100), textcolor=color.rgb(100, 100, 100), style=label.style_label_right, size=size.small)
        liveOBs.push(ActiveOB.new(b, l, imb_lbl, imb_line_t, imb_line_b, top, bot, imb_t, imb_b, has_imb, isBull, origin_bar_idx))

purge_ob(int threshold_idx) =>
    if liveOBs.size() > 0
        for i = liveOBs.size() - 1 to 0 
            ob = liveOBs.get(i)
            if ob.bar_idx < threshold_idx
                ob.box_id.delete()
                ob.lbl_id.delete()
                if ob.has_imb
                    ob.imb_label_id.delete()
                    ob.imb_line_top.delete()
                    ob.imb_line_bot.delete()
                liveOBs.remove(i)

check_single_ob(int offset, bool isBull, int current_limit_offset) =>
    bool found = false
    if offset + 1 < bar_index and offset - 1 >= current_limit_offset
        if isBull
            if close[offset+1] < open[offset+1] and low[offset] < low[offset+1] and high[offset+1] < low[offset-1]
                ob_top = high[offset+1], ob_bot = low[offset]
                imb_top = low[offset-1], imb_bot = high[offset+1]   
                mitigated = false
                for m = offset - 1 to current_limit_offset 
                    if low[m] <= ob_top
                        mitigated := true
                        break
                    if low[m] < imb_top
                        imb_top := low[m]
                if not mitigated
                    draw_ob(ob_top, ob_bot, true, bar_index - (offset + 1), imb_top, imb_bot)
                    found := true
        else
            if close[offset+1] > open[offset+1] and high[offset] > high[offset+1] and low[offset+1] > high[offset-1]
                ob_top = high[offset], ob_bot = low[offset+1]
                imb_top = low[offset+1], imb_bot = high[offset-1]   
                mitigated = false
                for m = offset - 1 to current_limit_offset 
                    if high[m] >= ob_bot
                        mitigated := true
                        break
                    if high[m] > imb_bot
                        imb_bot := high[m]
                if not mitigated
                    draw_ob(ob_top, ob_bot, false, bar_index - (offset + 1), imb_top, imb_bot)
                    found := true
    found

scan_origin_obs(int origin_idx, int break_idx, bool isBull) =>
    if EnableOB and origin_idx != -1 and origin_idx <= break_idx
        bars_back_origin = bar_index - origin_idx
        bars_back_break  = bar_index - break_idx
        if bars_back_origin > bars_back_break
            for j = bars_back_origin to bars_back_break
                check_single_ob(j, isBull, bars_back_break)

z_get_index_by_time(int target_t) =>
    int idx = -1
    int sz = array.size(z_a_t)
    if sz > 0
        for i = 0 to sz - 1
            if array.get(z_a_t, i) == target_t
                idx := i

    idx

z_add_sd_zone(int swing_idx, int bos_idx, bool isSellZone, float broken_level) =>
    if ShowZone
        int ext_idx = swing_idx
        float actual_broken_level = broken_level
        if bos_idx >= 0 and swing_idx >= bos_idx
            float extreme_val = isSellZone ? array.get(z_a_h, bos_idx) : array.get(z_a_l, bos_idx)
            ext_idx := bos_idx
            int count = swing_idx - bos_idx + 1
            for j = 0 to count - 1
                int check_idx = bos_idx + j
                if check_idx < array.size(z_a_h)
                    if isSellZone
                        if array.get(z_a_h, check_idx) >= extreme_val
                            extreme_val := array.get(z_a_h, check_idx)
                            ext_idx := check_idx
                    else
                        if array.get(z_a_l, check_idx) <= extreme_val
                            extreme_val := array.get(z_a_l, check_idx)
                            ext_idx := check_idx
                            
        if (na(actual_broken_level) or actual_broken_level <= 0) and bos_idx >= 0 and swing_idx > bos_idx
            int check_count = swing_idx - bos_idx
            float brk_val = isSellZone ? array.get(z_a_l, bos_idx + 1) : array.get(z_a_h, bos_idx + 1)
            for j = 1 to check_count
                int check_idx = bos_idx + j
                if check_idx < array.size(z_a_l)
                    if isSellZone
                        if array.get(z_a_l, check_idx) < brk_val
                            brk_val := array.get(z_a_l, check_idx)
                    else
                        if array.get(z_a_h, check_idx) > brk_val
                            brk_val := array.get(z_a_h, check_idx)
            actual_broken_level := brk_val
            
        int targetColor = isSellZone ? 0 : 1
        int k = ext_idx
        float zHigh = array.get(z_a_ha_h, ext_idx)
        float zLow  = array.get(z_a_ha_l, ext_idx)
        int total = array.size(z_a_ha_h)
        
        while k < total and (array.get(z_a_ha_c, k) >= array.get(z_a_ha_o, k) ? 0 : 1) != targetColor and k <= ext_idx + 10
            k += 1
            
        if k <= ext_idx + 10 and k < total
            if isSellZone
                zLow := array.get(z_a_ha_l, k)
                while k < total and (array.get(z_a_ha_c, k) >= array.get(z_a_ha_o, k) ? 0 : 1) == targetColor
                    zLow := math.min(zLow, array.get(z_a_ha_l, k))
                    k += 1
                zHigh := math.max(zHigh, array.get(z_a_ha_h, ext_idx))
            else
                zHigh := array.get(z_a_ha_h, k)
                while k < total and (array.get(z_a_ha_c, k) >= array.get(z_a_ha_o, k) ? 0 : 1) == targetColor
                    zHigh := math.max(zHigh, array.get(z_a_ha_h, k))
                    k += 1
                zLow := math.min(zLow, array.get(z_a_ha_l, ext_idx))
                
        if not na(actual_broken_level) and actual_broken_level > 0
            if isSellZone and zLow <= actual_broken_level + syminfo.mintick
                zHigh := array.get(z_a_ha_h, ext_idx)
                zLow := array.get(z_a_ha_l, ext_idx)
            else if not isSellZone and zHigh >= actual_broken_level - syminfo.mintick
                zHigh := array.get(z_a_ha_h, ext_idx)
                zLow := array.get(z_a_ha_l, ext_idx)
                
        int tStart = array.get(z_a_t, ext_idx)
        int tEnd = tStart + timeframe.in_seconds(timeframe.period) * 10 * 1000 
        box b = box.new(tStart, zHigh, tEnd, zLow, xloc=xloc.bar_time, border_color=na, bgcolor=isSellZone ? c_SellZone : c_BuyZone)
        
        if isSellZone
            sell_zones.push(SDZone.new(b, zLow, zHigh, true))
            if sell_zones.size() > MaxZones
                removed = sell_zones.shift()
                removed.box_id.delete()
        else
            buy_zones.push(SDZone.new(b, zHigh, zLow, false))
            if buy_zones.size() > MaxZones
                removed = buy_zones.shift()
                removed.box_id.delete()

z_clear_zones(bool isBuy) =>

    else

    if not na(active_kl_up_line)
        line.delete(active_kl_up_line), label.delete(active_kl_up_lbl)

    if z_ActiveLow_isActive and curr_ha_c < z_ActiveLow_price
        if not z_is_major_choch_dn
            z_last_break_dir := -1, z_last_break_type := 1, z_latest_break_dn_time := curr_t, z_latest_break_dn_lvl := z_ActiveLow_price
            
            // TÌM ĐỈNH CAO NHẤT TRONG NHỊP PULLBACK (DỜI PROTECTED HIGH)
            float pb_h = curr_h
            int pb_h_t = curr_t
            int lookback_idx = z_get_index_by_time(z_ActiveLow_time)
            if lookback_idx > 0
                for k = 0 to lookback_idx
                    if array.get(z_a_h, k) > pb_h
                        pb_h := array.get(z_a_h, k)
                        pb_h_t := array.get(z_a_t, k)
            z_maj_prot_h := pb_h
            z_maj_prot_h_time := pb_h_t

            int origin_idx = z_get_index_by_time(z_last_maj_h_time)
            if origin_idx != -1
                z_add_sd_zone(origin_idx, 0, true, z_ActiveLow_price)
        z_ActiveLow_isActive := false

// 8. CẬP NHẬT UI REAL-TIME (TRACKING ĐƯỜNG KIM CHỈ NAM & CÁC CẤP LEVEL)
// ==============================================================================
var line track_high_line = na, var label track_high_lbl = na
var line track_low_line = na, var label track_low_lbl = na

if barstate.islast
    int ext_t = time + timeframe.in_seconds(timeframe.period) * 10 * 1000 
    
    // Kéo dài các Box Zone ra hiện tại
    if buy_zones.size() > 0
        for i = 0 to buy_zones.size() - 1
            buy_zones.get(i).box_id.set_right(ext_t)
    if sell_zones.size() > 0
        for i = 0 to sell_zones.size() - 1
            sell_zones.get(i).box_id.set_right(ext_t)

    // VẼ ĐƯỜNG KIM CHỈ NAM (TRACKING LINES - CẤP PROTECTED & ACTIVE)
    // Priority giống MQL5: g_high = Protected High (downtrend) ưu tiên hơn Active High (fallback)
    //                       g_low  = Protected Low  (uptrend)  ưu tiên hơn Active Low  (fallback)
    if ShowTrackingLines
        float g_high_lvl = na, string g_high_text = "", int g_high_idx = na
        float g_low_lvl = na, string g_low_text = "", int g_low_idx = na

        // g_high: Protected High khi downtrend, fallback sang Active High
        if major_trend == -1 and not na(maj_prot_h)
            g_high_lvl := maj_prot_h, g_high_text := "Protected High", g_high_idx := maj_prot_h_idx
        else if active_high_is_active
            g_high_lvl := active_high_price, g_high_text := "Active High", g_high_idx := active_high_idx

        // g_low: Protected Low khi uptrend, fallback sang Active Low
        if major_trend == 1 and not na(maj_prot_l)
            g_low_lvl := maj_prot_l, g_low_text := "Protected Low", g_low_idx := maj_prot_l_idx
        else if active_low_is_active
            g_low_lvl := active_low_price, g_low_text := "Active Low", g_low_idx := active_low_idx

        // Cập nhật đường Trần (High) -> Đẩy text xuống DƯỚI (upper_left)
        if not na(track_high_line)
            track_high_line.delete(), track_high_lbl.delete()
        if not na(g_high_lvl) and not na(g_high_idx)
            track_high_line := line.new(g_high_idx, g_high_lvl, bar_index + 10, g_high_lvl, color=c_TrackingLine, style=line.style_dotted, extend=extend.right)
            track_high_lbl := label.new(bar_index + 5, g_high_lvl, g_high_text, color=color.new(color.white, 100), textcolor=c_TrackingLine, style=label.style_label_upper_left, size=size.small)

        // Cập nhật đường Sàn (Low) -> Đẩy text lên TRÊN (lower_left)
        if not na(track_low_line)
            track_low_line.delete(), track_low_lbl.delete()
        if not na(g_low_lvl) and not na(g_low_idx)
            track_low_line := line.new(g_low_idx, g_low_lvl, bar_index + 10, g_low_lvl, color=c_TrackingLine, style=line.style_dotted, extend=extend.right)
            track_low_lbl := label.new(bar_index + 5, g_low_lvl, g_low_text, color=color.new(color.white, 100), textcolor=c_TrackingLine, style=label.style_label_lower_left, size=size.small)

    // VẼ ĐƯỜNG WEAK LEVEL
    if ShowWeakLevel
        if not na(weak_level_line)
            line.delete(weak_level_line), label.delete(weak_level_lbl)
        
        if not na(weak_level_price) and not na(weak_level_idx)
            bool is_weak_high = (major_trend == 1)
            color w_col = is_weak_high ? c_WeakHigh : c_WeakLow
            string w_text = is_weak_high ? "Weak High" : "Weak Low"
            
            // is_weak_high (Uptrend): Text TRÊN -> lower_left
            // is_weak_low (Downtrend): Text DƯỚI -> upper_left
            weak_level_line := line.new(weak_level_idx, weak_level_price, bar_index + 10, weak_level_price, color=w_col, style=line.style_dashed, extend=extend.right)
            weak_level_lbl := label.new(bar_index + 5, weak_level_price, w_text, color=color.new(color.white, 100), textcolor=w_col, style=is_weak_high ? label.style_label_lower_left : label.style_label_upper_left, size=size.small)
