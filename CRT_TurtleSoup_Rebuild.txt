//@version=6
indicator("CRT | Turtle Soup (Rebuild - AnhTuan)", overlay=true, max_lines_count=500, max_labels_count=500)

// ==============================================================================
// BẢN DỰNG LẠI (REBUILD) - dùng để đối chiếu 1:1 với bản gốc "CRT | Turtle Soup (ICT)"
// Các input được đặt tên/nhóm bám sát bảng cấu hình gốc để dễ so sánh song song.
// Những chỗ còn GIẢ ĐỊNH (chưa xác nhận 100% qua video/ảnh gốc) được đánh dấu "// GIẢ ĐỊNH".
//
// THIẾT KẾ LẠI LẦN 4 (09/07) - MÔ HÌNH "ĐỈNH/ĐÁY TREO ĐỘC LẬP":
// Bằng chứng đối chiếu H4 (nến 08/07 13:00, Low=4041.62): TS-line và O-line của bản gốc
// KHÔNG LUÔN thuộc cùng 1 nến (TS khớp Low nến này, nhưng O lại cao hơn hẳn High nến này
// -> lấy từ 1 nến KHÁC, cũ hơn, chưa từng bị phá). Vậy TS và O phải được theo dõi ĐỘC LẬP:
//   - "Đỉnh treo" (pending high) = giá cao nhất chưa từng bị phá gần đây.
//   - "Đáy treo"  (pending low)  = giá thấp nhất chưa từng bị phá gần đây.
//   - Bearish xác nhận khi: 1 nến phá đỉnh treo rồi ĐÓNG CỬA quay lại dưới đỉnh treo đó.
//     TS = đỉnh treo vừa bị phá. O (target) = đáy treo HIỆN TẠI (của phía còn lại).
//   - Bullish xác nhận khi: 1 nến phá đáy treo rồi ĐÓNG CỬA quay lại trên đáy treo đó.
//     TS = đáy treo vừa bị phá. O (target) = đỉnh treo HIỆN TẠI (của phía còn lại).
// Case Daily xác nhận trước đây (Monday=A cho cả TS lẫn O) chỉ là TRÙNG HỢP: hôm đó cùng
// 1 nến vừa là đỉnh treo vừa là đáy treo (nến biên độ rộng, lập cực trị cả 2 phía).
//
// GHI CHÚ: input "Rangos TS a mostrar" (Primera/Última/Ambas) của bản gốc bị BỎ ở bản này -
// với mô hình đỉnh/đáy treo độc lập, khái niệm "Primera TS" (container cố định) không còn
// rõ ràng áp dụng được. Chỉ hiển thị range XÁC NHẬN GẦN NHẤT. Sẽ bổ sung lại nếu có thêm
// bằng chứng cần thiết.
// ==============================================================================

// ---------------- TEMPORALIDAD ----------------
grp_tf   = "TEMPORALIDAD"
useTF    = input.bool(true, "TF", group=grp_tf)
rangeTF  = input.timeframe("", "  Timeframe của range", group=grp_tf)

// ---------------- ESTILO ----------------
grp_style   = "ESTILO"
styleTheme  = input.string("TLAB colors", "Style",
     options=["TLAB colors", "standart colors", "Monochrome black BG", "Monochrome grey BG", "Custom"], group=grp_style)
customColor = input.color(color.new(#000000, 0), "Custom color", group=grp_style)
txtSizeStr  = input.string("small", "Tamaño Textos", options=["tiny", "small", "normal", "large"], group=grp_style)
lineExt     = input.int(20, "Extensión líneas", minval=0, group=grp_style)
objWidth    = input.int(1, "Ancho líneas OBJ", minval=1, group=grp_style)
tsWidth     = input.int(1, "Ancho líneas TS", minval=1, group=grp_style)

themeBaseColor() =>
    switch styleTheme
        "TLAB colors"          => color.new(#2962FF, 0)  // GIẢ ĐỊNH mã màu
        "standart colors"      => color.new(#787B86, 0)  // GIẢ ĐỊNH mã màu
        "Monochrome black BG"  => color.new(#FFFFFF, 0)  // GIẢ ĐỊNH mã màu
        "Monochrome grey BG"   => color.new(#000000, 0)  // GIẢ ĐỊNH mã màu
        "Custom"                => customColor
        => customColor

baseColor = themeBaseColor()
objColor  = baseColor
tsColor   = color.new(baseColor, 35)

// ---------------- CONFIGURACION TS ----------------
grp_ts        = "CONFIGURACION TS"
showOnlyObj   = input.bool(false, "Mostrar solo OBJ", group=grp_ts)
showPossible  = input.bool(true, "Posible TS al tomar High/Low", group=grp_ts)

// ---------------- ALERTA ----------------
grp_alert     = "ALERTA"
alertOnTarget = input.bool(true, "Alerta Obj alcanzado", group=grp_alert)

// ---------------- DEBUG (chỉ phục vụ đối chiếu, KHÔNG có trong bản gốc) ----------------
grp_dbg          = "DEBUG (đối chiếu)"
showTargetMarker = input.bool(false, "Hiện marker Obj Alcanzado (X)", group=grp_dbg)
showDebugTable   = input.bool(true, "Hiện bảng số liệu range gần nhất", group=grp_dbg)
waitForClose = input.bool(false, "Chỉ tính khi nến đã đóng cửa (an toàn, KHÁC bản gốc)", group=grp_dbg)

// ---------------- CALCULO ----------------
calcBars = input.int(1000, "Calculated bars", minval=10, group="CALCULO")

// ==============================================================================
// HELPERS
// ==============================================================================
labelSize() =>
    switch txtSizeStr
        "tiny"   => size.tiny
        "small"  => size.small
        "normal" => size.normal
        "large"  => size.large
        => size.normal

tfSuffix = useTF and rangeTF != "" ? rangeTF : timeframe.period

withinCalcWindow = (last_bar_index - bar_index) <= calcBars

crossedLevel(level) =>
    not na(level) and ((close[1] <= level and close > level) or (close[1] >= level and close < level))

// ==============================================================================
// LẤY DỮ LIỆU
// ==============================================================================
rH = useTF and rangeTF != "" ? request.security(syminfo.tickerid, rangeTF, high, lookahead=barmerge.lookahead_off) : high
rL = useTF and rangeTF != "" ? request.security(syminfo.tickerid, rangeTF, low, lookahead=barmerge.lookahead_off) : low
rC = useTF and rangeTF != "" ? request.security(syminfo.tickerid, rangeTF, close, lookahead=barmerge.lookahead_off) : close

curH = rH
curL = rL
curC = rC

// ==============================================================================
// THEO DÕI ĐỈNH/ĐÁY "TREO" ĐỘC LẬP + PHÁT HIỆN XÁC NHẬN
// ==============================================================================
var float pendHighVal  = na
var int   pendHighBar  = na
var int   pendHighTime = na
var float pendLowVal   = na
var int   pendLowBar   = na
var int   pendLowTime  = na

// Giữ lại giá trị TRƯỚC khi cập nhật, dùng để vẽ tick "Posible TS"
oldPendHighVal = pendHighVal
oldPendLowVal  = pendLowVal

bool  bearishConfirmed = false
bool  bullishConfirmed = false
float tsP     = na
float oP      = na
int   aBarIdx = na
int   aBarTime = na

// QUAN TRỌNG: phải giới hạn theo withinCalcWindow ở ngay đây (không chỉ ở phần hiển thị),
// nếu không pendHighVal/pendLowVal có thể "nhặt" phải 1 đỉnh/đáy treo từ RẤT XA trong quá
// khứ (ngoài phạm vi Calculated bars) rồi giữ mãi vì chưa từng bị phá lại -> ra giá vô lý
// (ví dụ Giá O = 251.90 dù giá hiện tại quanh 4000).
processBar = (not waitForClose or barstate.isconfirmed) and withinCalcWindow

if processBar
    // ---- Đỉnh treo (Bearish TS / Bullish O) ----
    if na(pendHighVal)
        pendHighVal := curH
        pendHighBar := bar_index
        pendHighTime := time
    else if curH > pendHighVal
        if curC < pendHighVal
            bearishConfirmed := true
            tsP := pendHighVal
            aBarIdx := pendHighBar
            aBarTime := pendHighTime
            pendHighVal := na
            pendHighBar := na
            pendHighTime := na
        else
            pendHighVal := curH
            pendHighBar := bar_index
            pendHighTime := time

    // ---- Đáy treo (Bullish TS / Bearish O) ----
    if na(pendLowVal)
        pendLowVal := curL
        pendLowBar := bar_index
        pendLowTime := time
    else if curL < pendLowVal
        if curC > pendLowVal
            bullishConfirmed := true
            tsP := pendLowVal
            aBarIdx := pendLowBar
            aBarTime := pendLowTime
            pendLowVal := na
            pendLowBar := na
            pendLowTime := na
        else
            pendLowVal := curL
            pendLowBar := bar_index
            pendLowTime := time

    // O (target) = đỉnh/đáy treo HIỆN TẠI của phía còn lại (sau khi đã cập nhật ở trên)
    // XÁC NHẬN qua case Daily (Giá O = 1628.73, 1 đáy CÓ THẬT nhưng từ ~3-4 năm trước, chưa
    // từng bị phá lại vì giá chỉ tăng liên tục kể từ đó): đáy/đỉnh treo phía đối diện, SAU KHI
    // đã dùng làm target, phải được LÀM MỚI LẠI TỪ ĐẦU - nếu không nó có thể "sống sót" qua
    // nhiều năm không hề liên quan tới vùng giá hiện tại.
    if bearishConfirmed
        oP := pendLowVal
        pendLowVal := na
        pendLowBar := na
        pendLowTime := na
    if bullishConfirmed
        oP := pendHighVal
        pendHighVal := na
        pendHighBar := na
        pendHighTime := na

bearishSweep = processBar and not na(oldPendHighVal) and curH > oldPendHighVal
bullishSweep = processBar and not na(oldPendLowVal) and curL < oldPendLowVal

// ==============================================================================
// TICK "POSIBLE TS" (cảnh báo tạm thời trên nến đang chạy - live, không lưu vĩnh viễn)
// ==============================================================================
var line possibleTsLine = na
if showPossible and withinCalcWindow and barstate.islast
    if not na(possibleTsLine)
        line.delete(possibleTsLine)
        possibleTsLine := na
    if bearishSweep
        possibleTsLine := line.new(bar_index[1], oldPendHighVal, bar_index, oldPendHighVal, xloc.bar_index,
             extend=extend.none, color=tsColor, style=line.style_dotted, width=1)
    if bullishSweep
        possibleTsLine := line.new(bar_index[1], oldPendLowVal, bar_index, oldPendLowVal, xloc.bar_index,
             extend=extend.none, color=tsColor, style=line.style_dotted, width=1)

// ==============================================================================
// HIỂN THỊ RANGE XÁC NHẬN GẦN NHẤT (chỉ giữ 1 range tại 1 thời điểm)
// ==============================================================================
var line  lastTsLine  = na
var line  lastObjLine = na
var label lastTsLabel = na
var label lastObjLabel= na
var float lastOPriceActive = na

var int    lastDirDbg     = 0    // 1 = Bullish, -1 = Bearish
var int    lastABarTime   = na
var float  lastTsPriceDbg = na
var float  lastOPriceDbg  = na

targetCrossedNow = not na(lastOPriceActive) and crossedLevel(lastOPriceActive)

// ---- Target đạt được -> dọn range đang hiển thị ----
if targetCrossedNow
    if not na(lastTsLine)
        line.delete(lastTsLine)
    if not na(lastObjLine)
        line.delete(lastObjLine)
    if not na(lastTsLabel)
        label.delete(lastTsLabel)
    if not na(lastObjLabel)
        label.delete(lastObjLabel)
    lastTsLine := na
    lastObjLine := na
    lastTsLabel := na
    lastObjLabel := na
    lastOPriceActive := na
    lastDirDbg := 0
    lastABarTime := na
    lastTsPriceDbg := na
    lastOPriceDbg := na

// ---- Range mới xác nhận -> thay thế range đang hiển thị ----
if (bearishConfirmed or bullishConfirmed) and withinCalcWindow
    if not na(lastTsLine)
        line.delete(lastTsLine)
    if not na(lastObjLine)
        line.delete(lastObjLine)
    if not na(lastTsLabel)
        label.delete(lastTsLabel)
    if not na(lastObjLabel)
        label.delete(lastObjLabel)

    tsLn = showOnlyObj ? na : line.new(aBarIdx, tsP, bar_index + lineExt, tsP, xloc.bar_index,
         extend=extend.none, color=tsColor, style=line.style_dashed, width=tsWidth)
    objLn = line.new(aBarIdx, oP, bar_index + lineExt, oP, xloc.bar_index,
         extend=extend.none, color=objColor, style=line.style_solid, width=objWidth)
    tsLb = showOnlyObj ? na : label.new(bar_index, tsP, "TS" + tfSuffix,
         style=label.style_label_upper_left, textcolor=tsColor, color=color.new(color.white, 100), size=labelSize())
    objLb = label.new(bar_index, oP, "O." + tfSuffix,
         style=label.style_label_lower_left, textcolor=objColor, color=color.new(color.white, 100), size=labelSize())

    lastTsLine  := tsLn
    lastObjLine := objLn
    lastTsLabel := tsLb
    lastObjLabel:= objLb
    lastOPriceActive := oP

    lastDirDbg     := bearishConfirmed ? -1 : 1
    lastABarTime   := aBarTime
    lastTsPriceDbg := tsP
    lastOPriceDbg  := oP

// ==============================================================================
// CẬP NHẬT ĐỘ DÀI LINE MỖI BAR - theo "Extensión líneas" (line luôn vươn tới bar_index + lineExt)
// ==============================================================================
if not na(lastTsLine)
    line.set_x2(lastTsLine, bar_index + lineExt)
if not na(lastObjLine)
    line.set_x2(lastObjLine, bar_index + lineExt)

// ==============================================================================
// ALERT - giá chạm vùng target (OBJ) của range gần nhất
// ==============================================================================
targetHit = alertOnTarget and targetCrossedNow
alertcondition(targetHit, title="CRT Obj alcanzado", message="CRT | Turtle Soup: giá đã chạm vùng target (OBJ)")

plotshape(showTargetMarker and targetHit, title="Obj Alcanzado", style=shape.xcross, location=location.abovebar, color=color.new(color.yellow, 0), size=size.tiny)

// ==============================================================================
// BẢNG DEBUG - số liệu chính xác của range gần nhất để đối chiếu với bản gốc
// ==============================================================================
var table dbgTable = table.new(position.top_right, 2, 4, border_width=1)
if showDebugTable and barstate.islast
    table.cell(dbgTable, 0, 0, "Hướng", bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 1, 0, lastDirDbg == 1 ? "Bullish" : lastDirDbg == -1 ? "Bearish" : "-", bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 0, 1, "Nến gốc (A)", bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 1, 1, na(lastABarTime) ? "-" : str.format_time(lastABarTime, "yyyy-MM-dd HH:mm", syminfo.timezone), bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 0, 2, "Giá TS", bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 1, 2, na(lastTsPriceDbg) ? "-" : str.tostring(lastTsPriceDbg, format.mintick), bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 0, 3, "Giá O (target)", bgcolor=color.gray, text_color=color.white)
    table.cell(dbgTable, 1, 3, na(lastOPriceDbg) ? "-" : str.tostring(lastOPriceDbg, format.mintick), bgcolor=color.gray, text_color=color.white)
